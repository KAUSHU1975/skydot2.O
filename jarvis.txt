<script>
/* =========================================================
   J.A.R.V.I.S. — Skydot voice-control layer (prototype)
   ---------------------------------------------------------
   Flow: user logs into Skydot -> JARVIS boots -> greets user
   by name -> listens/types -> answers questions AND routes
   recognised commands to a mock Skydot control panel.

   NOTE: login here is a client-side demo (accounts live only
   in this tab's memory) so the full experience can be shown
   end-to-end. A real product needs a real backend for auth,
   and the "Skydot panel" here is a stand-in for the actual
   Skydot app it would control via a shared API.
   ========================================================= */

const users = []; // {name, email, password} — in-memory only
let currentUser = null;
let authMode = 'signin';

/* ---------------- AUTH ---------------- */
function setAuthTab(mode){
  authMode = mode;
  document.getElementById('tabSignin').classList.toggle('active', mode==='signin');
  document.getElementById('tabSignup').classList.toggle('active', mode==='signup');
  document.getElementById('authSubmitBtn').textContent = mode==='signin' ? 'Sign In' : 'Create Account';
  document.getElementById('authError').textContent = '';
}
function submitAuth(){
  const name = document.getElementById('authName').value.trim();
  const email = document.getElementById('authEmail').value.trim().toLowerCase();
  const password = document.getElementById('authPassword').value;
  const errEl = document.getElementById('authError');
  errEl.textContent = '';

  if(!email || !password){ errEl.textContent = 'Email and password are required.'; return; }

  if(authMode === 'signup'){
    if(!name){ errEl.textContent = 'Please enter your name.'; return; }
    if(users.find(u=>u.email===email)){ errEl.textContent = 'Account already exists — try Sign In.'; return; }
    users.push({name, email, password});
    currentUser = {name, email};
  } else {
    const u = users.find(u=>u.email===email && u.password===password);
    if(!u){ errEl.textContent = 'No matching account. Try Create Account.'; return; }
    currentUser = {name: u.name, email: u.email};
  }
  document.getElementById('loginScreen').classList.remove('on');
  runBoot();
}
function logout(){
  currentUser = null;
  document.getElementById('mainScreen').classList.remove('on');
  document.getElementById('authName').value='';
  document.getElementById('authEmail').value='';
  document.getElementById('authPassword').value='';
  document.getElementById('loginScreen').classList.add('on');
  window.speechSynthesis && window.speechSynthesis.cancel();
}

/* ---------------- BOOT SEQUENCE ---------------- */
const BOOT_LINES = [
  'Initializing core systems...',
  'Linking to Skydot workspace...',
  'Calibrating voice interface...',
  'Loading user profile...',
  'All systems online.'
];
function runBoot(){
  document.getElementById('bootScreen').classList.add('on');
  const log = document.getElementById('bootLog');
  let i = 0;
  const iv = setInterval(()=>{
    log.textContent = BOOT_LINES[i];
    i++;
    if(i >= BOOT_LINES.length){
      clearInterval(iv);
      setTimeout(()=>{
        document.getElementById('bootScreen').classList.remove('on');
        startMain();
      }, 500);
    }
  }, 420);
}

}

}

/* ---------------- COMMAND ROUTER ---------------- */
function routeCommand(text){
  const t = text.toLowerCase();
  if(/\b(file|upload|cloud)\b/.test(t) && /(open|show|go to)/.test(t)){ setSection('files'); return `Opening My Files, ${currentUser.name}.`; }
  if(/\bspace/.test(t) && /(open|show|go to)/.test(t)){ setSection('spaces'); return `Opening Spaces now.`; }
  if(/\b(chat|talk)\b/.test(t) && /(open|show|go to)/.test(t)){ setSection('chat'); return `Opening AI Chat.`; }
  if(/(quiz|studio|summary|summarize|questions|notes)\b/.test(t)){ setSection('studio'); return `Opening AI Studio — I can generate that once you point me to a file or some text.`; }
  if(/\b(home|dashboard)\b/.test(t) && /(open|show|go to)/.test(t)){ setSection('home'); return `Back to Home.`; }
  return null;
}

/* ---------------- AI CALL (works inside Claude.ai's artifact preview) ---------------- */
async function askJarvis(userText){
  try{
    const response = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({
        model: "claude-sonnet-4-6",
        max_tokens: 500,
        system: `You are J.A.R.V.I.S., a calm, witty, highly capable AI voice assistant, in the voice-control layer of an app called Skydot. The current user is named ${currentUser.name}. Address them naturally and occasionally use "sir" or their name, Jarvis-style. Keep answers conversational and not too long, since they may be read aloud.`,
        messages: [{role:'user', content:userText}]
      })
    });
    if(!response.ok) throw new Error('API error '+response.status);
    const data = await response.json();
    return (data.content||[]).map(b=>b.type==='text'?b.text:'').join('\n').trim() || "I don't have a response for that, sir.";
  }catch(err){
    console.error(err);
    return "I can't reach my reasoning core right now. Outside Claude.ai's preview, J.A.R.V.I.S. needs its own backend and API key wired in to answer questions.";
  }
}

/* ---------------- MESSAGE HANDLING ---------------- */
async function handleUserText(raw){
  const text = (raw||'').trim();
  if(!text) return;
  addLine('user', text);

  const commandReply = routeCommand(text);
  if(commandReply){
    addLine('jarvis', commandReply);
    speak(commandReply);
    return;
  }

  setStatus('Thinking...');
  const reply = await askJarvis(text);
  setStatus('Standing by, sir.');
  addLine('jarvis', reply);
  speak(reply);
}
function setStatus(s){ document.getElementById('jarvisStatus').textContent = s; }

/* ---------------- VOICE: recognition + synthesis ---------------- */
let recognition = null;
let listening = false;
function getRecognition(){
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  if(!SR) return null;
  const r = new SR();
  r.lang = 'en-IN'; r.interimResults = false; r.maxAlternatives = 1;
  return r;
}
function toggleMic(){
  if(listening){ recognition && recognition.stop(); return; }
  recognition = getRecognition();
  if(!recognition){ setStatus('Speech recognition not supported in this browser.'); return; }
  listening = true;
  document.getElementById('orb').classList.add('listening');
  document.getElementById('micBtn').classList.add('on');
  setStatus('Listening...');
  recognition.onresult = (e) => {
    const said = e.results[0][0].transcript;
    handleUserText(said);
  };
  recognition.onerror = () => { setStatus('Did not catch that, sir.'); };
  recognition.onend = () => {
    listening = false;
    document.getElementById('orb').classList.remove('listening');
    document.getElementById('micBtn').classList.remove('on');
    if(document.getElementById('jarvisStatus').textContent === 'Listening...') setStatus('Standing by, sir.');
  };
  recognition.start();
}
function speak(text){
  if(!('speechSynthesis' in window)) return;
  window.speechSynthesis.cancel();
  const u = new SpeechSynthesisUtterance(text);
  const voices = window.speechSynthesis.getVoices();
  const preferred = voices.find(v=>/male|david|daniel|google uk english male/i.test(v.name)) || voices[0];
  if(preferred) u.voice = preferred;
  u.rate = 1.0; u.pitch = 0.95;
  u.onstart = () => document.getElementById('orb').classList.add('speaking');
  u.onend = () => document.getElementById('orb').classList.remove('speaking');
  window.speechSynthesis.speak(u);
}
function stopSpeaking(){ window.speechSynthesis && window.speechSynthesis.cancel(); document.getElementById('orb').classList.remove('speaking'); }
function populateVoicesQuiet(){
  if(!window.speechSynthesis) return;
  window.speechSynthesis.getVoices();
  window.speechSynthesis.onvoiceschanged = () => window.speechSynthesis.getVoices();
}

/* ---------------- BACKGROUND RADAR FIELD ---------------- */
function drawBg(){
  const c = document.getElementById('bg');
  const ctx = c.getContext('2d');
  function resize(){ c.width = innerWidth; c.height = innerHeight; }
  resize(); window.addEventListener('resize', resize);
  const N = Math.min(60, Math.floor(innerWidth/26));
  const pts = Array.from({length:N}, ()=>({
    x: Math.random()*innerWidth, y: Math.random()*innerHeight,
    vx: (Math.random()-0.5)*0.12, vy: (Math.random()-0.5)*0.12
  }));
  function frame(){
    ctx.clearRect(0,0,c.width,c.height);
    ctx.fillStyle = 'rgba(77,217,236,0.5)';
    ctx.strokeStyle = 'rgba(77,217,236,0.12)';
    for(const p of pts){
      p.x += p.vx; p.y += p.vy;
      if(p.x<0||p.x>innerWidth) p.vx*=-1;
      if(p.y<0||p.y>innerHeight) p.vy*=-1;
      ctx.beginPath(); ctx.arc(p.x,p.y,1.3,0,7); ctx.fill();
    }
    for(let i=0;i<pts.length;i++){
      for(let j=i+1;j<pts.length;j++){
        const dx=pts[i].x-pts[j].x, dy=pts[i].y-pts[j].y, d=Math.hypot(dx,dy);
        if(d<110){ ctx.globalAlpha = 1-d/110; ctx.beginPath(); ctx.moveTo(pts[i].x,pts[i].y); ctx.lineTo(pts[j].x,pts[j].y); ctx.stroke(); ctx.globalAlpha=1; }
      }
    }
    requestAnimationFrame(frame);
  }
  frame();
}
drawBg();
